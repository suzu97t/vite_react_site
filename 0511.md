"use client";

import { useState } from "react";

// ============================================================
// 型定義
// ============================================================
type Step = "step1" | "step2" | "step3" | "step4";

interface Message {
  question: string;
  answer: string;
}

interface Viewpoint {
  id: string;
  text: string;
  isEditing: boolean;
}

interface ExtractQuestion {
  id: string;
  text: string;
}

// ============================================================
// モックデータ生成関数
// ============================================================
const generateViewpoints = (answers: string[]): Viewpoint[] => [
  { id: "v1", text: "外観の色・艶・変色の確認", isEditing: false },
  { id: "v2", text: "触感による硬度・弾力性の確認", isEditing: false },
  { id: "v3", text: "においによる腐敗・酸化の確認", isEditing: false },
  { id: "v4", text: "音（打音・摩擦音）による内部状態の確認", isEditing: false },
  { id: "v5", text: "温度・湿度の変化への着目", isEditing: false },
];

const generateExtractQuestions = (viewpoint: string): ExtractQuestion[] => [
  { id: "q1", text: `${viewpoint}において、最初に確認すべきポイントはどこですか？` },
  { id: "q2", text: `${viewpoint}で「異常だ」と判断する基準は何ですか？` },
  { id: "q3", text: `${viewpoint}に関して、新人が見落としやすい点はどこですか？` },
  { id: "q4", text: `${viewpoint}で過去に失敗した経験やヒヤリハットはありますか？` },
];

const generateFollowUpQuestion = (
  viewpoint: string,
  previousQuestion: string,
  answer: string,
  depth: number
): string => {
  const followUps = [
    `なぜ「${answer.slice(0, 15)}...」と判断できるのですか？その根拠を教えてください。`,
    `その判断をするために、どのくらいの経験・期間が必要でしたか？`,
    `具体的にどのような状態になったら、次のアクションを取りますか？`,
    `その見極めを数値や言葉で表現するとしたら、どのように表しますか？`,
    `同じ判断を他の人に教えるとしたら、どのように伝えますか？`,
  ];
  return followUps[depth % followUps.length];
};

// ============================================================
// サブコンポーネント
// ============================================================

/** テキストエリア付き質問カード */
function QuestionCard({
  question,
  onAnswer,
  extraActions,
}: {
  question: string;
  onAnswer: (answer: string) => void;
  extraActions?: React.ReactNode;
}) {
  const [input, setInput] = useState("");

  const handleSubmit = () => {
    if (!input.trim()) return;
    onAnswer(input.trim());
    setInput("");
  };

  return (
    <div className="bg-blue-50 border border-blue-200 rounded-xl p-4 space-y-3">
      <p className="text-sm font-semibold text-blue-800">🤖 AI</p>
      <p className="text-gray-800">{question}</p>
      <textarea
        className="w-full border border-gray-300 rounded-lg p-2 text-sm resize-none focus:outline-none focus:ring-2 focus:ring-blue-400"
        rows={4}
        placeholder="回答を入力してください..."
        value={input}
        onChange={(e) => setInput(e.target.value)}
      />
      <div className="flex flex-wrap gap-2">
        <button
          onClick={handleSubmit}
          className="bg-blue-600 text-white px-4 py-2 rounded-lg text-sm hover:bg-blue-700 transition"
        >
          回答する
        </button>
        {extraActions}
      </div>
    </div>
  );
}

/** 着眼点カード */
function ViewpointCard({
  viewpoint,
  onSelect,
  onEdit,
  onEditConfirm,
  onEditChange,
}: {
  viewpoint: Viewpoint;
  onSelect: (v: Viewpoint) => void;
  onEdit: (id: string) => void;
  onEditConfirm: (id: string) => void;
  onEditChange: (id: string, text: string) => void;
}) {
  return (
    <div className="border border-gray-200 rounded-xl p-3 bg-white shadow-sm flex items-center justify-between gap-2">
      {viewpoint.isEditing ? (
        <input
          className="flex-1 border border-blue-300 rounded-lg px-2 py-1 text-sm focus:outline-none focus:ring-2 focus:ring-blue-400"
          value={viewpoint.text}
          onChange={(e) => onEditChange(viewpoint.id, e.target.value)}
          autoFocus
        />
      ) : (
        <span className="flex-1 text-sm text-gray-800">{viewpoint.text}</span>
      )}
      <div className="flex gap-1">
        {viewpoint.isEditing ? (
          <button
            onClick={() => onEditConfirm(viewpoint.id)}
            className="text-xs bg-green-500 text-white px-2 py-1 rounded hover:bg-green-600 transition"
          >
            確定
          </button>
        ) : (
          <>
            <button
              onClick={() => onEdit(viewpoint.id)}
              className="text-xs bg-gray-100 text-gray-600 px-2 py-1 rounded hover:bg-gray-200 transition"
            >
              編集
            </button>
            <button
              onClick={() => onSelect(viewpoint)}
              className="text-xs bg-blue-600 text-white px-2 py-1 rounded hover:bg-blue-700 transition"
            >
              選択
            </button>
          </>
        )}
      </div>
    </div>
  );
}

/** 引き出し質問カード */
function ExtractQuestionCard({
  question,
  onSelect,
}: {
  question: ExtractQuestion;
  onSelect: (q: ExtractQuestion) => void;
}) {
  return (
    <div
      onClick={() => onSelect(question)}
      className="border border-gray-200 rounded-xl p-3 bg-white shadow-sm cursor-pointer hover:bg-blue-50 hover:border-blue-300 transition"
    >
      <span className="text-sm text-gray-800">{question.text}</span>
    </div>
  );
}

/** ノウハウ記入欄 */
function KnowledgePanel({
  disabled,
  selectedViewpoint,
  knowledgeText,
  onKnowledgeChange,
}: {
  disabled: boolean;
  selectedViewpoint: string | null;
  knowledgeText: string;
  onKnowledgeChange: (text: string) => void;
}) {
  return (
    <div className="h-full flex flex-col bg-white border border-gray-200 rounded-xl p-4 space-y-3">
      <h2 className="text-base font-bold text-gray-700 border-b pb-2">📝 ノウハウ記入欄</h2>

      {selectedViewpoint && (
        <div className="bg-yellow-50 border border-yellow-200 rounded-lg p-2">
          <p className="text-xs text-yellow-700 font-semibold">着眼点</p>
          <p className="text-sm text-gray-800 mt-1">{selectedViewpoint}</p>
        </div>
      )}

      {disabled ? (
        <div className="flex-1 bg-gray-50 rounded-lg border border-dashed border-gray-300 flex items-center justify-center">
          <p className="text-gray-400 text-sm">着眼点を選択後に記入できます</p>
        </div>
      ) : (
        <textarea
          className="flex-1 w-full border border-gray-300 rounded-lg p-3 text-sm resize-none focus:outline-none focus:ring-2 focus:ring-yellow-400"
          placeholder="ここにノウハウを自由に記入・整理してください..."
          value={knowledgeText}
          onChange={(e) => onKnowledgeChange(e.target.value)}
          rows={20}
        />
      )}
    </div>
  );
}

// ============================================================
// メインコンポーネント
// ============================================================
export default function HomePage() {
  const [step, setStep] = useState<Step>("step1");
  const [step1Phase, setStep1Phase] = useState<0 | 1 | 2 | 3>(0); // 0=Q1, 1=Q2, 2=Q3, 3=done

  // STEP1
  const [step1Answers, setStep1Answers] = useState<string[]>([]);

  // STEP2
  const [viewpoints, setViewpoints] = useState<Viewpoint[]>([]);
  const [newViewpointText, setNewViewpointText] = useState("");

  // STEP3
  const [selectedViewpoint, setSelectedViewpoint] = useState<Viewpoint | null>(null);
  const [extractQuestions, setExtractQuestions] = useState<ExtractQuestion[]>([]);

  // STEP4
  const [selectedExtractQuestion, setSelectedExtractQuestion] =
    useState<ExtractQuestion | null>(null);
  const [currentQuestion, setCurrentQuestion] = useState<string>("");
  const [deepDiveDepth, setDeepDiveDepth] = useState(0);
  const [conversationHistory, setConversationHistory] = useState<Message[]>([]);

  // ノウハウ
  const [knowledgeText, setKnowledgeText] = useState("");

  // ============================================================
  // STEP1 ハンドラ
  // ============================================================
  const step1Questions = [
    "五感をどのように使って何を感じていますか？",
    "他に注意するポイントや着眼点はありますか？",
    "こういった観点で注意していることや着眼点はありますか？",
  ];

  const handleStep1Answer = (answer: string) => {
    const newAnswers = [...step1Answers, answer];
    setStep1Answers(newAnswers);

    if (step1Phase < 2) {
      setStep1Phase((step1Phase + 1) as 0 | 1 | 2 | 3);
    } else {
      // 3問完了 → STEP2へ
      const generated = generateViewpoints(newAnswers);
      setViewpoints(generated);
      setStep("step2");
    }
  };

  // ============================================================
  // STEP2 ハンドラ
  // ============================================================
  const handleSelectViewpoint = (viewpoint: Viewpoint) => {
    setSelectedViewpoint(viewpoint);
    const questions = generateExtractQuestions(viewpoint.text);
    setExtractQuestions(questions);
    setKnowledgeText("");
    setConversationHistory([]);
    setDeepDiveDepth(0);
    setStep("step3");
  };

  const handleEditViewpoint = (id: string) => {
    setViewpoints((prev) =>
      prev.map((v) => (v.id === id ? { ...v, isEditing: true } : v))
    );
  };

  const handleEditConfirm = (id: string) => {
    setViewpoints((prev) =>
      prev.map((v) => (v.id === id ? { ...v, isEditing: false } : v))
    );
  };

  const handleEditChange = (id: string, text: string) => {
    setViewpoints((prev) => prev.map((v) => (v.id === id ? { ...v, text } : v)));
  };

  const handleAddViewpoint = () => {
    if (!newViewpointText.trim()) return;
    const newV: Viewpoint = {
      id: `v${Date.now()}`,
      text: newViewpointText.trim(),
      isEditing: false,
    };
    setViewpoints((prev) => [...prev, newV]);
    setNewViewpointText("");
  };

  // ============================================================
  // STEP3 ハンドラ
  // ============================================================
  const handleSelectExtractQuestion = (question: ExtractQuestion) => {
    setSelectedExtractQuestion(question);
    setCurrentQuestion(question.text);
    setDeepDiveDepth(0);
    setConversationHistory([]);
    setStep("step4");
  };

  // ============================================================
  // STEP4 ハンドラ
  // ============================================================
  const handleDeepDiveAnswer = (answer: string) => {
    const newHistory: Message[] = [
      ...conversationHistory,
      { question: currentQuestion, answer },
    ];
    setConversationHistory(newHistory);

    const nextDepth = deepDiveDepth + 1;
    setDeepDiveDepth(nextDepth);

    const nextQuestion = generateFollowUpQuestion(
      selectedViewpoint?.text ?? "",
      currentQuestion,
      answer,
      nextDepth
    );
    setCurrentQuestion(nextQuestion);
  };

  const handleBackToStep2 = () => {
    setSelectedViewpoint(null);
    setExtractQuestions([]);
    setSelectedExtractQuestion(null);
    setCurrentQuestion("");
    setConversationHistory([]);
    setDeepDiveDepth(0);
    setStep("step2");
  };

  const handleBackToStep3 = () => {
    setSelectedExtractQuestion(null);
    setCurrentQuestion("");
    setConversationHistory([]);
    setDeepDiveDepth(0);
    setStep("step3");
  };

  // ============================================================
  // レンダリング - 左カラム
  // ============================================================
  const renderLeftColumn = () => {
    // STEP1
    if (step === "step1") {
      return (
        <div className="space-y-4">
          {/* 回答済み質問の表示 */}
          {step1Answers.map((ans, i) => (
            <div key={i} className="space-y-2">
              <div className="bg-blue-50 border border-blue-200 rounded-xl p-3">
                <p className="text-xs font-semibold text-blue-700 mb-1">🤖 AI</p>
                <p className="text-sm text-gray-700">{step1Questions[i]}</p>
              </div>
              <div className="bg-gray-50 border border-gray-200 rounded-xl p-3 ml-4">
                <p className="text-xs font-semibold text-gray-500 mb-1">👤 あなた</p>
                <p className="text-sm text-gray-800">{ans}</p>
              </div>
            </div>
          ))}

          {/* 現在の質問 */}
          {step1Phase <= 2 && (
            <QuestionCard
              question={step1Questions[step1Phase]}
              onAnswer={handleStep1Answer}
            />
          )}
        </div>
      );
    }

    // STEP2
    if (step === "step2") {
      return (
        <div className="space-y-4">
          <div className="bg-blue-50 border border-blue-200 rounded-xl p-4">
            <p className="text-sm font-semibold text-blue-800 mb-2">🤖 AI</p>
            <p className="text-gray-800">
              深堀りしたい着眼点を選んでください
            </p>
          </div>

          <div className="space-y-2">
            {viewpoints.map((v) => (
              <ViewpointCard
                key={v.id}
                viewpoint={v}
                onSelect={handleSelectViewpoint}
                onEdit={handleEditViewpoint}
                onEditConfirm={handleEditConfirm}
                onEditChange={handleEditChange}
              />
            ))}
          </div>

          {/* 着眼点追加 */}
          <div className="flex gap-2 mt-2">
            <input
              className="flex-1 border border-gray-300 rounded-lg px-3 py-2 text-sm focus:outline-none focus:ring-2 focus:ring-blue-400"
              placeholder="着眼点を追加..."
              value={newViewpointText}
              onChange={(e) => setNewViewpointText(e.target.value)}
              onKeyDown={(e) => e.key === "Enter" && handleAddViewpoint()}
            />
            <button
              onClick={handleAddViewpoint}
              className="bg-blue-100 text-blue-700 px-3 py-2 rounded-lg text-sm hover:bg-blue-200 transition"
            >
              ＋ 追加
            </button>
          </div>
        </div>
      );
    }

    // STEP3
    if (step === "step3") {
      return (
        <div className="space-y-4">
          <div className="bg-blue-50 border border-blue-200 rounded-xl p-4">
            <p className="text-sm font-semibold text-blue-800 mb-2">🤖 AI</p>
            <p className="text-gray-800">
              <span className="font-semibold text-blue-700">
                「{selectedViewpoint?.text}」
              </span>
              について深堀りします。
              <br />
              インタビュー質問を選んでください。
            </p>
          </div>

          <button
            onClick={handleBackToStep2}
            className="text-xs text-gray-500 underline hover:text-gray-700 transition"
          >
            ← 着眼点を選びなおす
          </button>

          <div className="space-y-2">
            {extractQuestions.map((q) => (
              <ExtractQuestionCard
                key={q.id}
                question={q}
                onSelect={handleSelectExtractQuestion}
              />
            ))}
          </div>
        </div>
      );
    }

    // STEP4
    if (step === "step4") {
      return (
        <div className="space-y-4">
          {/* ラベル */}
          <div className="bg-blue-50 border border-blue-200 rounded-xl p-3">
            <p className="text-xs text-blue-600 font-semibold">
              🔍 深掘り中：{selectedViewpoint?.text}
            </p>
          </div>

          {/* 会話履歴 */}
          {conversationHistory.map((msg, i) => (
            <div key={i} className="space-y-2">
              <div className="bg-blue-50 border border-blue-100 rounded-xl p-3">
                <p className="text-xs font-semibold text-blue-700 mb-1">🤖 AI</p>
                <p className="text-sm text-gray-700">{msg.question}</p>
              </div>
              <div className="bg-gray-50 border border-gray-200 rounded-xl p-3 ml-4">
                <p className="text-xs font-semibold text-gray-500 mb-1">👤 あなた</p>
                <p className="text-sm text-gray-800">{msg.answer}</p>
              </div>
            </div>
          ))}

          {/* 現在の質問 */}
          <QuestionCard
            question={currentQuestion}
            onAnswer={handleDeepDiveAnswer}
            extraActions={
              <div className="flex flex-wrap gap-2">
                <button
                  onClick={handleBackToStep2}
                  className="text-xs text-gray-500 underline hover:text-gray-700 transition"
                >
                  着眼点を選びなおす
                </button>
                <button
                  onClick={handleBackToStep3}
                  className="text-xs text-gray-500 underline hover:text-gray-700 transition"
                >
                  インタビュー質問を選びなおす
                </button>
              </div>
            }
          />
        </div>
      );
    }
  };

  // ============================================================
  // レンダリング - メイン
  // ============================================================
  return (
    <div className="min-h-screen bg-gray-100">
      {/* ヘッダー */}
      <header className="bg-white border-b border-gray-200 px-6 py-3 flex items-center gap-3">
        <span className="text-xl">🧠</span>
        <h1 className="text-lg font-bold text-gray-800">暗黙知抽出アプリ</h1>
        {/* ステップインジケーター */}
        <div className="ml-auto flex items-center gap-2 text-xs">
          {[
            { key: "step1", label: "観点洗い出し" },
            { key: "step2", label: "着眼点選択" },
            { key: "step3", label: "質問選択" },
            { key: "step4", label: "深掘り対話" },
          ].map((s, i, arr) => (
            <div key={s.key} className="flex items-center gap-1">
              <span
                className={`px-2 py-1 rounded-full font-semibold ${
                  step === s.key
                    ? "bg-blue-600 text-white"
                    : "bg-gray-200 text-gray-500"
                }`}
              >
                {i + 1}. {s.label}
              </span>
              {i < arr.length - 1 && (
                <span className="text-gray-400">→</span>
              )}
            </div>
          ))}
        </div>
      </header>

      {/* 2カラムレイアウト */}
      <div className="flex h-[calc(100vh-56px)]">
        {/* 左カラム：AI質問エリア */}
        <div className="w-1/2 border-r border-gray-200 overflow-y-auto p-6 space-y-4">
          <h2 className="text-sm font-bold text-gray-500 uppercase tracking-wider mb-4">
            AI 質問
          </h2>
          {renderLeftColumn()}
        </div>

        {/* 右カラム：ノウハウ記入欄 */}
        <div className="w-1/2 overflow-y-auto p-6">
          <KnowledgePanel
            disabled={step === "step1" || step === "step2"}
            selectedViewpoint={selectedViewpoint?.text ?? null}
            knowledgeText={knowledgeText}
            onKnowledgeChange={setKnowledgeText}
          />
        </div>
      </div>
    </div>
  );
}
